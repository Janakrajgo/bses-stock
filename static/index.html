// [Compound Annual Growth Rate (CAGR) %]
Long Term CAGR % = 
VAR FirstDate = MIN('BSE_Stock_Data'[Date])
VAR LastDate = MAX('BSE_Stock_Data'[Date])
VAR InitialPrice = CALCULATE(MIN('BSE_Stock_Data'[Close_Price]), 'BSE_Stock_Data'[Date] = FirstDate)
VAR FinalPrice = CALCULATE(MAX('BSE_Stock_Data'[Close_Price]), 'BSE_Stock_Data'[Date] = LastDate)
VAR NumYears = DIVIDE(DATEDIFF(FirstDate, LastDate, DAY), 365.25, 1)
RETURN
    IF(NumYears > 0 && InitialPrice > 0, ((FinalPrice / InitialPrice) ^ (1 / NumYears)) - 1, BLANK())

// [Sharpe Ratio (Indian Repo Rf = 6.5%)]
Sharpe Ratio = 
VAR RiskFreeRateDaily = 0.065 / 252
VAR AvgDailyReturn = AVERAGE('BSE_Stock_Data'[Daily_Return_Pct])
VAR DailyStd = STDEV.S('BSE_Stock_Data'[Daily_Return_Pct])
RETURN
    DIVIDE((AvgDailyReturn - RiskFreeRateDaily) * SQRT(252), DailyStd * SQRT(252), 0)
